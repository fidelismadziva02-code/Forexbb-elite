import React, { useRef } from "react";

  View, Text, ScrollView, TouchableOpacity,
  StyleSheet, FlatList,
} from "react-native";
import { useTradeLog, type LogEntry } from "@/contexts/BotContext-FINAL";

function logColor(type: LogEntry["type"]): string {
  switch (type) {
    case "TRADE_OPEN":  return "#00C896"; // green
    case "TRADE_CLOSE": return "#FF6B6B"; // red
    case "SIGNAL":      return "#4FC3F7"; // blue
    case "CONNECT":     return "#FFD54F"; // yellow
    case "ERROR":       return "#FF5252"; // bright red
    default:            return "#B0BEC5"; // grey
  }
}

function logIcon(type: LogEntry["type"]): string {
  switch (type) {
    case "TRADE_OPEN":  return "▲";
    case "TRADE_CLOSE": return "▼";
    case "SIGNAL":      return "◆";
    case "CONNECT":     return "⬡";
    case "ERROR":       return "✕";
    default:            return "·";
  }
}

function formatTime(ts: number): string {
  return new Date(ts).toLocaleTimeString("en-GB", { hour12: false });
}

function LogRow({ item }: { item: LogEntry }) {
  const color = logColor(item.type);
  return (
    <View style={styles.row}>
      <Text style={[styles.icon, { color }]}>{logIcon(item.type)}</Text>
      <Text style={styles.time}>{formatTime(item.timestamp)}</Text>
      {item.pair && (
        <Text style={[styles.pair, { color }]}>{item.pair}</Text>
      )}
      <Text style={styles.msg} numberOfLines={2}>{item.message}</Text>
    </View>
  );
}

export function TradeLog() {
  const { logs, clearLog } = useTradeLog();
  const flatRef = useRef<FlatList>(null);

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <View style={styles.headerLeft}>
          <View style={styles.dot} />
          <Text style={styles.headerTitle}>LIVE BOARD LOG</Text>
        </View>
        <View style={styles.headerRight}>
          <Text style={styles.count}>{logs.length} entries</Text>
          <TouchableOpacity onPress={clearLog} style={styles.clearBtn}>
            <Text style={styles.clearText}>CLEAR</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* Legend */}
      <View style={styles.legend}>
        {[
          { type: "TRADE_OPEN"  as const, label: "Open"   },
          { type: "TRADE_CLOSE" as const, label: "Close"  },
          { type: "SIGNAL"      as const, label: "Signal" },
          { type: "CONNECT"     as const, label: "Connect"},
          { type: "ERROR"       as const, label: "Error"  },
        ].map(({ type, label }) => (
          <View key={type} style={styles.legendItem}>
            <Text style={[styles.legendIcon, { color: logColor(type) }]}>{logIcon(type)}</Text>
            <Text style={styles.legendLabel}>{label}</Text>
          </View>
        ))}
      </View>

      {/* Log entries */}
      {logs.length === 0 ? (
        <View style={styles.empty}>
          <Text style={styles.emptyText}>Waiting for bot activity...</Text>
          <Text style={styles.emptySubtext}>Connect MT5 and start the bot to see live logs here.</Text>
        </View>
      ) : (
        <FlatList
          ref={flatRef}
          data={logs}
          keyExtractor={item => item.id}
          renderItem={({ item }) => <LogRow item={item} />}
          style={styles.list}
          contentContainerStyle={styles.listContent}
          showsVerticalScrollIndicator={false}
          onContentSizeChange={() => flatRef.current?.scrollToOffset({ offset: 0, animated: false })}
        />
      )}
    </View>
  );
}
